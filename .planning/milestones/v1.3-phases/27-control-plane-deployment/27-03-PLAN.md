  <action>
Create `helm/safedns/templates/ui-deployment.yaml` based on the existing `k8s/base/ui-deployment.yaml` but using Helm templating. Include:

1. **Deployment**:
   ```yaml
   apiVersion: apps/v1
   kind: Deployment
   metadata:
     name: {{ include "safedns.ui.fullname" . }}
     namespace: {{ .Release.Namespace }}
     labels:
       {{- include "safedns.labels" . | nindent 4 }}
       app.kubernetes.io/component: ui
   spec:
     replicas: {{ .Values.ui.replicaCount }}
     selector:
       matchLabels:
         {{- include "safedns.selectorLabels" . | nindent 6 }}
         app.kubernetes.io/component: ui
     strategy:
       type: RollingUpdate
       rollingUpdate:
         maxSurge: 1
         maxUnavailable: 0
     template:
       metadata:
         labels:
           {{- include "safedns.selectorLabels" . | nindent 8 }}
           app.kubernetes.io/component: ui
       spec:
         securityContext:
           runAsNonRoot: true
           runAsUser: 1001
           fsGroup: 1001
         containers:
           - name: ui
             image: "{{ .Values.ui.image.repository }}:{{ .Values.ui.image.tag | default .Chart.AppVersion }}"
             imagePullPolicy: {{ .Values.ui.image.pullPolicy }}
             ports:
               - name: http
                 containerPort: 3000
                 protocol: TCP
             env:
               - name: NODE_ENV
                 value: "production"
               - name: PORT
                 value: "3000"
               - name: HOSTNAME
                 value: "0.0.0.0"
               # Runtime env vars (NOT NEXT_PUBLIC_)
               - name: API_URL
                 value: {{ .Values.ui.env.API_URL | quote }}
               - name: APP_URL
                 value: {{ .Values.ui.env.APP_URL | quote }}
               - name: ADMIN_UI_URL
                 value: {{ .Values.ui.env.ADMIN_UI_URL | quote }}
               # Stripe publishable key (build-time, but safe to pass as env too)
               {{- if .Values.ui.env.STRIPE_PUBLISHABLE_KEY }}
               - name: NEXT_PUBLIC_STRIPE_PUBLISHABLE_KEY
                 value: {{ .Values.ui.env.STRIPE_PUBLISHABLE_KEY | quote }}
               {{- end }}
             resources:
               {{- toYaml .Values.ui.resources | nindent 14 }}
             livenessProbe:
               httpGet:
                 path: /api/config
                 port: http
               initialDelaySeconds: 30
               periodSeconds: 10
               timeoutSeconds: 5
               failureThreshold: 3
             readinessProbe:
               httpGet:
                 path: /api/config
                 port: http
               initialDelaySeconds: 5
               periodSeconds: 5
               timeoutSeconds: 3
               failureThreshold: 3
             securityContext:
               allowPrivilegeEscalation: false
               readOnlyRootFilesystem: false  # Next.js standalone needs write access to .next
               capabilities:
                 drop:
                   - ALL
         affinity:
           podAntiAffinity:
             preferredDuringSchedulingIgnoredDuringExecution:
               - weight: 100
                 podAffinityTerm:
                   labelSelector:
                     matchLabels:
                       {{- include "safedns.selectorLabels" . | nindent 22 }}
                       app.kubernetes.io/component: ui
                   topologyKey: kubernetes.io/hostname
   ```

2. **Service**:
   ```yaml
   ---
   apiVersion: v1
   kind: Service
   metadata:
     name: {{ include "safedns.ui.fullname" . }}
     namespace: {{ .Release.Namespace }}
     labels:
       {{- include "safedns.labels" . | nindent 4 }}
       app.kubernetes.io/component: ui
   spec:
     type: ClusterIP
     ports:
       - port: 80
         targetPort: http
         protocol: TCP
         name: http
     selector:
       {{- include "safedns.selectorLabels" . | nindent 4 }}
       app.kubernetes.io/component: ui
   ```

NOTE: Use UID 1001 (matching the Dockerfile's nextjs user). The liveness probe uses `/api/config` which is the route created in Plan 27-02. Service port 80 maps to container port 3000 for clean Ingress routing.
