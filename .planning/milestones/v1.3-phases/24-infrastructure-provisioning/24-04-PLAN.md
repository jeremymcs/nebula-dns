   ```bash
   #!/usr/bin/env bash
   # Nebula DNS - Create Kubernetes Secrets from Terraform Outputs
   # Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>
   #
   # Reads database credentials from Terraform outputs and creates
   # Kubernetes Secrets in the nebula-dns namespace.
   #
   # Prerequisites:
   #   - kubectl configured with DOKS cluster context
   #   - Terraform applied in infrastructure/terraform/environments/production/
   #   - nebula-dns namespace exists
   #
   # Usage: ./create-secrets.sh [--dry-run]

   set -euo pipefail

   SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
   TF_DIR="${SCRIPT_DIR}/../../terraform/environments/production"
   NAMESPACE="nebula-dns"
   DRY_RUN=""
   APPLY_CMD="kubectl apply"

   if [[ "${1:-}" == "--dry-run" ]]; then
     DRY_RUN="--dry-run=client"
     echo "DRY RUN MODE - no changes will be applied"
   fi

   # Validate prerequisites
   command -v kubectl >/dev/null 2>&1 || { echo "kubectl required"; exit 1; }
   command -v terraform >/dev/null 2>&1 || { echo "terraform required"; exit 1; }

   # Ensure namespace exists
   kubectl create namespace "${NAMESPACE}" ${DRY_RUN} 2>/dev/null || true

   echo "=== Reading Terraform Outputs ==="
   cd "${TF_DIR}"

   # PostgreSQL
   PG_URI=$(terraform output -raw postgres_connection_uri 2>/dev/null || echo "")
   PG_DIRECT_URI=$(terraform output -raw postgres_direct_uri 2>/dev/null || echo "")

   # Valkey
   VALKEY_URI=$(terraform output -raw valkey_connection_uri 2>/dev/null || echo "")

   # ClickHouse (endpoints from ClickHouse Cloud)
   CH_ENDPOINTS=$(terraform output -json clickhouse_endpoints 2>/dev/null || echo "{}")

   cd "${SCRIPT_DIR}"

   # Validate we got values
   if [[ -z "${PG_URI}" ]]; then
     echo "WARNING: postgres_connection_uri is empty. Have you run 'terraform apply'?"
     echo "You can also set secrets manually using the template:"
     echo "  cat ${SCRIPT_DIR}/secret-template.yaml"
     echo ""
   fi

   echo "=== Creating Kubernetes Secrets ==="

   # --- API Secrets ---
   echo "Creating api-secrets..."
   kubectl create secret generic api-secrets \
     --namespace "${NAMESPACE}" \
     --from-literal=database-url="${PG_URI}" \
     --from-literal=database-direct-url="${PG_DIRECT_URI}" \
     --from-literal=redis-url="${VALKEY_URI}" \
     --from-literal=jwt-secret="$(openssl rand -base64 48)" \
     --from-literal=jwt-refresh-secret="$(openssl rand -base64 48)" \
     --from-literal=admin-jwt-secret="$(openssl rand -base64 48)" \
     --from-literal=mfa-encryption-key="$(openssl rand -hex 32)" \
     --from-literal=sso-encryption-key="$(openssl rand -hex 32)" \
     --from-literal=siem-encryption-key="$(openssl rand -hex 32)" \
     ${DRY_RUN} -o yaml | ${APPLY_CMD} ${DRY_RUN} -f -

   echo "api-secrets created"

   # --- Policy Service Secrets ---
   echo "Creating policy-svc-secrets..."
   kubectl create secret generic policy-svc-secrets \
     --namespace "${NAMESPACE}" \
     --from-literal=redis-url="${VALKEY_URI}" \
     --from-literal=clickhouse-url="${CH_ENDPOINTS}" \
     ${DRY_RUN} -o yaml | ${APPLY_CMD} ${DRY_RUN} -f -

   echo "policy-svc-secrets created"

   # --- ClickHouse Secrets ---
   echo "Creating clickhouse-secrets..."
   kubectl create secret generic clickhouse-secrets \
     --namespace "${NAMESPACE}" \
     --from-literal=username="default" \
     --from-literal=password="" \
     --from-literal=host="" \
     --from-literal=port="8443" \
     ${DRY_RUN} -o yaml | ${APPLY_CMD} ${DRY_RUN} -f -

   echo "clickhouse-secrets created (update username/password/host manually from ClickHouse Cloud console)"

   # --- External Service Secrets (manual) ---
   echo ""
   echo "=== Manual Secrets Required ==="
   echo "The following secrets must be set manually:"
   echo ""
   echo "  # Stripe (from Stripe Dashboard)"
   echo "  kubectl -n ${NAMESPACE} create secret generic stripe-secrets \\"
   echo "    --from-literal=secret-key='sk_...' \\"
   echo "    --from-literal=webhook-secret='whsec_...'"
   echo ""
   echo "  # SendGrid (from SendGrid Dashboard)"
   echo "  kubectl -n ${NAMESPACE} create secret generic sendgrid-secrets \\"
   echo "    --from-literal=api-key='SG.xxx'"
   echo ""
   echo "  # Sentry (from Sentry Dashboard)"
   echo "  kubectl -n ${NAMESPACE} create secret generic sentry-secrets \\"
   echo "    --from-literal=dsn='https://xxx@sentry.io/xxx'"
   echo ""
   echo "=== Done ==="
   ```

   ```bash
   #!/usr/bin/env bash
   # Nebula DNS - Infrastructure Verification
   # Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>
   #
   # Verifies that all provisioned infrastructure is accessible and healthy.
   # Run after terraform apply and operator installation.
   #
   # Usage: ./verify-infrastructure.sh

   set -euo pipefail

   PASS=0
   FAIL=0
   WARN=0

   check() {
     local name="$1"
     local cmd="$2"
     printf "%-45s" "Checking ${name}..."
     if eval "${cmd}" >/dev/null 2>&1; then
       echo "PASS"
       PASS=$((PASS + 1))
     else
       echo "FAIL"
       FAIL=$((FAIL + 1))
     fi
   }

   warn_check() {
     local name="$1"
     local cmd="$2"
     printf "%-45s" "Checking ${name}..."
     if eval "${cmd}" >/dev/null 2>&1; then
       echo "PASS"
       PASS=$((PASS + 1))
     else
       echo "WARN"
       WARN=$((WARN + 1))
     fi
   }

   echo "=== Nebula DNS Infrastructure Verification ==="
   echo ""

   # --- Kubernetes Cluster ---
   echo "--- DOKS Cluster ---"
   check "kubectl connectivity" "kubectl cluster-info"
   check "nodes ready (3+)" "test \$(kubectl get nodes --no-headers | grep -c Ready) -ge 3"
   check "K8s version 1.30+" "kubectl version --short 2>/dev/null | grep -E 'Server.*1\.(3[0-9]|[4-9][0-9])' || kubectl version -o json | grep -E 'gitVersion.*v1\.(3[0-9]|[4-9][0-9])'"
   check "nebula-dns namespace" "kubectl get namespace nebula-dns"

   echo ""

   # --- Kubernetes Operators ---
   echo "--- Operators ---"
   check "cert-manager pods running" "kubectl get pods -n cert-manager --no-headers | grep -c Running | grep -qE '^[3-9]'"
   check "cert-manager webhook ready" "kubectl get deployment cert-manager-webhook -n cert-manager -o jsonpath='{.status.readyReplicas}' | grep -qE '[1-9]'"
   check "ClusterIssuer letsencrypt-prod" "kubectl get clusterissuer letsencrypt-prod"
   check "external-dns running" "kubectl get pods -n kube-system -l app.kubernetes.io/name=external-dns --no-headers | grep Running"

   echo ""

   # --- Kubernetes Secrets ---
   echo "--- Secrets ---"
   check "api-secrets exists" "kubectl get secret api-secrets -n nebula-dns"
   check "policy-svc-secrets exists" "kubectl get secret policy-svc-secrets -n nebula-dns"
   check "clickhouse-secrets exists" "kubectl get secret clickhouse-secrets -n nebula-dns"
   warn_check "digitalocean-credentials exists" "kubectl get secret digitalocean-credentials -n kube-system"
   warn_check "digitalocean-dns exists" "kubectl get secret digitalocean-dns -n cert-manager"

   echo ""

   # --- Database Connectivity (from within cluster) ---
   echo "--- Database Connectivity ---"
   echo "(These checks require port-forwarding or VPC access)"
   warn_check "PostgreSQL secret has database-url" "kubectl get secret api-secrets -n nebula-dns -o jsonpath='{.data.database-url}' | base64 -d | grep -q postgresql"
   warn_check "Valkey secret has redis-url" "kubectl get secret api-secrets -n nebula-dns -o jsonpath='{.data.redis-url}' | base64 -d | grep -q rediss"
   warn_check "ClickHouse secret has host" "kubectl get secret clickhouse-secrets -n nebula-dns -o jsonpath='{.data.host}' | base64 -d | grep -q '.'"

   echo ""

   # --- Summary ---
   echo "=== Summary ==="
   echo "  PASS: ${PASS}"
   echo "  FAIL: ${FAIL}"
   echo "  WARN: ${WARN}"
   echo ""

   if [[ ${FAIL} -gt 0 ]]; then
     echo "RESULT: FAILED - ${FAIL} check(s) failed"
     exit 1
   elif [[ ${WARN} -gt 0 ]]; then
     echo "RESULT: PASSED with warnings"
     exit 0
   else
     echo "RESULT: ALL PASSED"
     exit 0
   fi
   ```

2. Create `infrastructure/scripts/rotate-credentials.sh`:
   ```bash
   #!/usr/bin/env bash
   # Nebula DNS - Credential Rotation Runbook
   # Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>
   #
   # Step-by-step procedures for rotating credentials.
   # Each rotation follows: generate new -> update secret -> restart pods -> verify
   #
   # Usage: ./rotate-credentials.sh <credential-type>
   #
   # Credential types:
   #   jwt          - Rotate JWT signing secrets
   #   encryption   - Rotate MFA/SSO/SIEM encryption keys
   #   postgres     - Rotate PostgreSQL password (via DigitalOcean)
   #   valkey       - Rotate Valkey password (via DigitalOcean)
   #   clickhouse   - Rotate ClickHouse password (via ClickHouse Cloud)
   #   do-token     - Rotate DigitalOcean API token
   #   all-app      - Rotate all application secrets (jwt + encryption)

   set -euo pipefail

   NAMESPACE="nebula-dns"

   usage() {
     echo "Usage: $0 <credential-type>"
     echo ""
     echo "Credential types:"
     echo "  jwt          Rotate JWT signing secrets (access, refresh, admin)"
     echo "  encryption   Rotate encryption keys (MFA, SSO, SIEM)"
     echo "  postgres     Rotate PostgreSQL password"
     echo "  valkey       Rotate Valkey password"
     echo "  clickhouse   Rotate ClickHouse Cloud password"
     echo "  do-token     Rotate DigitalOcean API token"
     echo "  all-app      Rotate all application-level secrets"
     echo ""
     echo "WARNING: Database credential rotation requires coordinated restart."
     echo "         Schedule during maintenance window."
     exit 1
   }

   rotate_jwt() {
     echo "=== Rotating JWT Secrets ==="
     echo ""
     echo "IMPACT: All active sessions will be invalidated."
     echo "DOWNTIME: Brief (pod restart ~30s)"
     echo ""
     read -p "Continue? (y/N) " -n 1 -r
     echo
     [[ $REPLY =~ ^[Yy]$ ]] || exit 0

     echo "Generating new JWT secrets..."
     NEW_JWT=$(openssl rand -base64 48)
     NEW_REFRESH=$(openssl rand -base64 48)
     NEW_ADMIN=$(openssl rand -base64 48)

     echo "Updating api-secrets..."
     kubectl get secret api-secrets -n "${NAMESPACE}" -o json \
       | jq --arg jwt "$(echo -n "$NEW_JWT" | base64)" \
            --arg refresh "$(echo -n "$NEW_REFRESH" | base64)" \
            --arg admin "$(echo -n "$NEW_ADMIN" | base64)" \
            '.data["jwt-secret"]=$jwt | .data["jwt-refresh-secret"]=$refresh | .data["admin-jwt-secret"]=$admin' \
       | kubectl apply -f -

     echo "Restarting API pods..."
     kubectl rollout restart deployment/api -n "${NAMESPACE}"
     kubectl rollout status deployment/api -n "${NAMESPACE}" --timeout=120s

     echo "JWT secrets rotated successfully."
     echo "NOTE: All users will need to re-authenticate."
   }

   rotate_encryption() {
     echo "=== Rotating Encryption Keys ==="
     echo ""
     echo "WARNING: This will make existing encrypted data (MFA secrets, SSO configs,"
     echo "         SIEM credentials) unreadable. You must re-encrypt data first."
     echo ""
     echo "STEPS:"
     echo "  1. Run encryption migration script: npm run migrate-encryption"
     echo "  2. Then rotate the keys"
     echo "  3. Restart API pods"
     echo ""
     read -p "Have you run the encryption migration? (y/N) " -n 1 -r
     echo
     [[ $REPLY =~ ^[Yy]$ ]] || { echo "Run migration first, then retry."; exit 1; }

     NEW_MFA=$(openssl rand -hex 32)
     NEW_SSO=$(openssl rand -hex 32)
     NEW_SIEM=$(openssl rand -hex 32)

     kubectl get secret api-secrets -n "${NAMESPACE}" -o json \
       | jq --arg mfa "$(echo -n "$NEW_MFA" | base64)" \
            --arg sso "$(echo -n "$NEW_SSO" | base64)" \
            --arg siem "$(echo -n "$NEW_SIEM" | base64)" \
            '.data["mfa-encryption-key"]=$mfa | .data["sso-encryption-key"]=$sso | .data["siem-encryption-key"]=$siem' \
       | kubectl apply -f -

     kubectl rollout restart deployment/api -n "${NAMESPACE}"
     kubectl rollout status deployment/api -n "${NAMESPACE}" --timeout=120s

     echo "Encryption keys rotated successfully."
   }

   rotate_postgres() {
     echo "=== Rotating PostgreSQL Password ==="
     echo ""
     echo "IMPACT: API will lose database connectivity during rotation."
     echo "DOWNTIME: ~1-2 minutes"
     echo "REQUIRES: doctl CLI authenticated"
     echo ""
     echo "STEPS:"
     echo "  1. Reset password in DigitalOcean:"
     echo "     doctl databases user reset <cluster-id> nebula"
     echo "  2. Get new connection URI from Terraform or DO Console"
     echo "  3. Update Kubernetes secret:"
     echo "     kubectl -n ${NAMESPACE} create secret generic api-secrets \\"
     echo "       --from-literal=database-url='NEW_URI' \\"
     echo "       --dry-run=client -o yaml | kubectl apply -f -"
     echo "  4. Update policy-svc-secrets if applicable"
     echo "  5. Restart affected pods:"
     echo "     kubectl rollout restart deployment/api -n ${NAMESPACE}"
     echo "  6. Verify connectivity:"
     echo "     kubectl logs deployment/api -n ${NAMESPACE} --tail=20"
   }

   rotate_valkey() {
     echo "=== Rotating Valkey Password ==="
     echo ""
     echo "IMPACT: Cache will be temporarily unavailable."
     echo "DOWNTIME: ~1-2 minutes"
     echo ""
     echo "STEPS:"
     echo "  1. Reset password in DigitalOcean:"
     echo "     doctl databases user reset <cluster-id> default"
     echo "  2. Get new connection URI"
     echo "  3. Update Kubernetes secrets (api-secrets AND policy-svc-secrets):"
     echo "     Update redis-url in both secrets"
     echo "  4. Restart affected pods:"
     echo "     kubectl rollout restart deployment/api -n ${NAMESPACE}"
     echo "     kubectl rollout restart deployment/policy-svc -n ${NAMESPACE}"
     echo "  5. Verify pub/sub connectivity:"
     echo "     kubectl logs deployment/policy-svc -n ${NAMESPACE} --tail=20"
   }

   rotate_clickhouse() {
     echo "=== Rotating ClickHouse Cloud Password ==="
     echo ""
     echo "STEPS:"
     echo "  1. Go to ClickHouse Cloud Console -> Service -> Settings -> Reset Password"
     echo "  2. Update clickhouse-secrets:"
     echo "     kubectl -n ${NAMESPACE} create secret generic clickhouse-secrets \\"
     echo "       --from-literal=username='default' \\"
     echo "       --from-literal=password='NEW_PASSWORD' \\"
     echo "       --from-literal=host='HOST' \\"
     echo "       --from-literal=port='8443' \\"
     echo "       --dry-run=client -o yaml | kubectl apply -f -"
     echo "  3. Update policy-svc-secrets with new clickhouse-url"
     echo "  4. Restart policy-svc:"
     echo "     kubectl rollout restart deployment/policy-svc -n ${NAMESPACE}"
   }

   rotate_do_token() {
     echo "=== Rotating DigitalOcean API Token ==="
     echo ""
     echo "STEPS:"
     echo "  1. Create new token in DO Console -> API -> Tokens"
     echo "  2. Update Terraform: export DIGITALOCEAN_TOKEN=new_token"
     echo "  3. Update cert-manager secret:"
     echo "     kubectl -n cert-manager create secret generic digitalocean-dns \\"
     echo "       --from-literal=access-token='NEW_TOKEN' \\"
     echo "       --dry-run=client -o yaml | kubectl apply -f -"
     echo "  4. Update external-dns secret:"
     echo "     kubectl -n kube-system create secret generic digitalocean-credentials \\"
     echo "       --from-literal=access-token='NEW_TOKEN' \\"
     echo "       --dry-run=client -o yaml | kubectl apply -f -"
     echo "  5. Restart cert-manager and external-dns:"
     echo "     kubectl rollout restart deployment/cert-manager -n cert-manager"
     echo "     kubectl rollout restart deployment/external-dns -n kube-system"
     echo "  6. Revoke old token in DO Console"
   }

   rotate_all_app() {
     rotate_jwt
     echo ""
     rotate_encryption
   }

   case "${1:-}" in
     jwt) rotate_jwt ;;
     encryption) rotate_encryption ;;
     postgres) rotate_postgres ;;
     valkey) rotate_valkey ;;
     clickhouse) rotate_clickhouse ;;
     do-token) rotate_do_token ;;
     all-app) rotate_all_app ;;
     *) usage ;;
   esac
   ```
   Make executable.
