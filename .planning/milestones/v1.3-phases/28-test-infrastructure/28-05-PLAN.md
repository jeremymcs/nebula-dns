  <action>
    Create `.github/workflows/load-test.yml`:

    ```yaml
    # Nebula DNS - Load Test Workflow
    # Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>
    #
    # Runs k6 load tests against DOKS staging endpoints

    name: Load Test

    on:
      schedule:
        - cron: '0 2 * * 1'  # Every Monday at 2 AM UTC
      workflow_dispatch:
        inputs:
          scenario:
            description: 'Test scenario to run'
            required: false
            default: 'smoke'
            type: choice
            options:
              - smoke
              - load
              - stress
          target:
            description: 'Target environment'
            required: false
            default: 'staging'
            type: choice
            options:
              - staging
              - local

    env:
      K6_VERSION: 'v0.54.0'

    jobs:
      api-load-test:
        name: API Load Test
        runs-on: ubuntu-latest

        steps:
          - name: Checkout code
            uses: actions/checkout@v4

          - name: Install k6
            run: |
              curl -L https://github.com/grafana/k6/releases/download/${{ env.K6_VERSION }}/k6-${{ env.K6_VERSION }}-linux-amd64.tar.gz | tar xvz
              sudo mv k6-${{ env.K6_VERSION }}-linux-amd64/k6 /usr/local/bin/
              k6 version

          - name: Run API smoke test
            run: |
              k6 run \
                --out json=api-results.json \
                --tag testid=api-smoke-$(date +%s) \
                scripts/load-testing/api-load-test.js
            env:
              API_URL: ${{ github.event.inputs.target == 'local' && 'http://localhost:4010' || secrets.STAGING_API_URL }}
              TEST_EMAIL: ${{ secrets.STAGING_TEST_EMAIL }}
              TEST_PASSWORD: ${{ secrets.STAGING_TEST_PASSWORD }}
              K6_SCENARIO: ${{ github.event.inputs.scenario || 'smoke' }}
            continue-on-error: true

          - name: Upload API results
            uses: actions/upload-artifact@v4
            if: always()
            with:
              name: k6-api-results-${{ github.run_id }}
              path: api-results.json
              retention-days: 30

      dns-load-test:
        name: DNS Load Test
        runs-on: ubuntu-latest

        steps:
          - name: Checkout code
            uses: actions/checkout@v4

          - name: Install k6
            run: |
              curl -L https://github.com/grafana/k6/releases/download/${{ env.K6_VERSION }}/k6-${{ env.K6_VERSION }}-linux-amd64.tar.gz | tar xvz
              sudo mv k6-${{ env.K6_VERSION }}-linux-amd64/k6 /usr/local/bin/
              k6 version

          - name: Run DNS smoke test
            run: |
              k6 run \
                --out json=dns-results.json \
                --tag testid=dns-smoke-$(date +%s) \
                scripts/load-testing/dns-load-test.js
            env:
              DOH_URL: ${{ github.event.inputs.target == 'local' && 'https://localhost:4443/dns-query' || secrets.STAGING_DOH_URL }}
              DNS_SERVER: ${{ secrets.STAGING_DNS_SERVER }}
              DNS_PORT: ${{ secrets.STAGING_DNS_PORT }}
              K6_SCENARIO: ${{ github.event.inputs.scenario || 'smoke' }}
            continue-on-error: true

          - name: Upload DNS results
            uses: actions/upload-artifact@v4
            if: always()
            with:
              name: k6-dns-results-${{ github.run_id }}
              path: dns-results.json
              retention-days: 30
    ```

    Key design decisions:
    - `continue-on-error: true` — load tests should not block CI, they are informational
    - Scheduled weekly on Mondays for baseline tracking
    - `workflow_dispatch` with scenario selector for on-demand testing
    - Both API and DNS tests run in parallel (separate jobs)
    - Results uploaded as artifacts for 30 days
    - k6 version pinned to v0.54.0
    - Staging secrets referenced but not created (those are manual GitHub Settings)
  </action>
