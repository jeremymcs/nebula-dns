            },
            want: Block,
        },

            policy: &Policy{
                BlockedCategories: []string{"malware", "phishing"},

            },
            want: Block,
        },
        // Add more test cases...
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {

        })
    }
}
