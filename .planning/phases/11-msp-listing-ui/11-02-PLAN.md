3. Add infinite scroll state and logic:
   ```tsx
   // Accumulated MSPs from all pages
   const [allMsps, setAllMsps] = useState<MspTenant[]>([])
   const [cursor, setCursor] = useState<string | undefined>(undefined)
   const [hasMore, setHasMore] = useState(true)
   const sentinelRef = useRef<HTMLDivElement>(null)
   const loadingMoreRef = useRef(false)

   // Fetch current page
   const { data, isLoading, error, mutate } = useAdminMspList({
     q: debouncedSearch || undefined,
     status: statusFilter || undefined,
     limit: 25,
     cursor,
   })

   // Reset when search/filter changes
   useEffect(() => {
     setAllMsps([])
     setCursor(undefined)
     setHasMore(true)
     loadingMoreRef.current = false
   }, [debouncedSearch, statusFilter])

   // Append new data
   useEffect(() => {
     if (data?.data) {
       if (cursor) {
         // Appending to existing
         setAllMsps(prev => [...prev, ...data.data])
       } else {
         // Fresh data
         setAllMsps(data.data)
       }
       setHasMore(data.pagination.hasMore)
       loadingMoreRef.current = false
     }
   }, [data, cursor])

   // Load more function
   const loadMore = useCallback(() => {
     if (loadingMoreRef.current || !hasMore || isLoading || !data?.pagination.cursor) return
     loadingMoreRef.current = true
     setCursor(data.pagination.cursor)
   }, [hasMore, isLoading, data?.pagination.cursor])

   // Intersection observer for infinite scroll
   useEffect(() => {
     const observer = new IntersectionObserver(
       (entries) => {
         if (entries[0].isIntersecting && hasMore && !isLoading) {
           loadMore()
         }
       },
       { threshold: 0.1 }
     )

     if (sentinelRef.current) {
       observer.observe(sentinelRef.current)
     }

     return () => observer.disconnect()
   }, [hasMore, isLoading, loadMore])
   ```

       if (menuRef.current && !menuRef.current.contains(event.target as Node)) {
