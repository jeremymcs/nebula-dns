   ```typescript
     private configService: ConfigService,
     private auditLogsService: AuditLogsService,
   ) {}
   ```

3. Change `canActivate` to `async canActivate` (returns Promise<boolean>) because it will call async audit logging.

4. After getting the user from request, check the flags BEFORE throwing ForbiddenException:

   ```typescript
   const rbacEnabled = this.configService.get<boolean>('admin.rbacEnabled', false);
   const shadowMode = this.configService.get<boolean>('admin.rbacShadowMode', true);

   // Check if role would be denied
   const wouldBeDenied = !user?.role || !requiredRoles.includes(user.role);

   if (wouldBeDenied) {
     // Log the would-be denial
     await this.logRbacDecision(user, request, requiredRoles, rbacEnabled && !shadowMode);

     // If RBAC is disabled, allow
     if (!rbacEnabled) {
       return true;
     }

     // If shadow mode is enabled, log but allow
     if (shadowMode) {
       this.logger.warn(
         `Shadow mode: Would deny ${user?.email || 'unknown'} access to ${request.method} ${request.url} (requires ${requiredRoles.join(' or ')})`
       );
       return true;
     }

     // RBAC enabled and shadow mode off - actually deny
     throw new ForbiddenException(`Requires ${requiredRoles.join(' or ')} role`);
   }

   return true;
   ```

5. Add private async method for audit logging:
   ```typescript
   private async logRbacDecision(
     user: any,
     request: any,
     requiredRoles: AdminRole[],
     isActualDenial: boolean,
   ): Promise<void> {
     const ADMIN_TENANT_ID = '00000000-0000-0000-0000-000000000000';

     try {
       await this.auditLogsService.log({
         tenantId: ADMIN_TENANT_ID,
         userId: null,
         action: isActualDenial ? 'rbac_denied' : 'rbac_shadow_denied',
         resourceType: 'admin',
         resourceId: user?.adminId || null,
         resourceName: user?.email || 'unknown',
         metadata: {
           method: request.method,
           path: request.url,
           requiredRoles,
           userRole: user?.role || null,
           shadowMode: !isActualDenial,
         },
       });
     } catch (error) {
       this.logger.error('Failed to log RBAC decision', error);
     }
   }
   ```

6. Update the existing role checks to use the new logic. Remove the inline ForbiddenException throws for role checks and use the new flag-aware logic instead.

Note: The synthetic tenant ID `00000000-0000-0000-0000-000000000000` is used for admin audit logs per project conventions (see STATE.md decision).
