Note: ClickHouse Cloud has an official Terraform provider (`ClickHouse/clickhouse`). Use it to create a service.

1. `variables.tf`:
   - `service_name` (string) - ClickHouse Cloud service name
   - `cloud_provider` (string, default "aws") - ClickHouse Cloud infra provider (aws, gcp, azure)
   - `region` (string, default "us-east-1") - Cloud region for ClickHouse service
   - `tier` (string, default "development") - Service tier (development, production)
   - `min_total_memory_gb` (number, default 24) - Minimum memory in GB
   - `max_total_memory_gb` (number, default 48) - Maximum memory in GB
   - `idle_scaling` (bool, default true) - Allow scaling to zero when idle
   - `idle_timeout_minutes` (number, default 5) - Minutes before scaling to zero

2. `main.tf`:
   ```hcl
   terraform {
     required_providers {
       clickhouse = {
         source  = "ClickHouse/clickhouse"
         version = "~> 0.2"
       }
     }
   }

   resource "clickhouse_service" "analytics" {
     name           = var.service_name
     cloud_provider = var.cloud_provider
     region         = var.region
     tier           = var.tier

     ip_access = [
       {
         source      = "0.0.0.0/0"
         description = "Allow all (restrict after DOKS egress IP is known)"
       }
     ]

     min_total_memory_gb  = var.min_total_memory_gb
     max_total_memory_gb  = var.max_total_memory_gb
     idle_scaling         = var.idle_scaling
     idle_timeout_minutes = var.idle_timeout_minutes
   }
   ```
   Note: IP access starts permissive; will be restricted to DOKS egress IPs in a later phase after cluster is running and we know the NAT gateway IPs.

   # Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>

   provider: digitalocean

   # Use environment variable for DO token (set via secret)
   env:
     - name: DO_TOKEN
       valueFrom:
         secretKeyRef:
           name: digitalocean-credentials
           key: access-token

   # Only manage our domain
   domainFilters:
     - nebuladns.io

   # Ownership tracking
   txtOwnerId: nebula-dns-prod
   txtPrefix: "_externaldns."

   # Sync mode: create, update, and delete records

   # Sources to watch

3. Create `infrastructure/kubernetes/operators/install-operators.sh`:
   ```bash
   #!/usr/bin/env bash
   # Nebula DNS - Kubernetes Operator Installation
   # Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>
   #
   # Installs cert-manager and external-dns on a DOKS cluster.
   # Prerequisites: kubectl configured, helm installed, DO_TOKEN set.
   #
   # Usage: ./install-operators.sh [--dry-run]

   set -euo pipefail

   SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
   DRY_RUN=""

   if [[ "${1:-}" == "--dry-run" ]]; then
     DRY_RUN="--dry-run"
     echo "DRY RUN MODE - no changes will be applied"
   fi

   # Validate prerequisites
   command -v kubectl >/dev/null 2>&1 || { echo "kubectl required but not found"; exit 1; }
   command -v helm >/dev/null 2>&1 || { echo "helm required but not found"; exit 1; }

   if [[ -z "${DO_TOKEN:-}" ]]; then
     echo "ERROR: DO_TOKEN environment variable must be set"
     echo "Get it from: DigitalOcean Control Panel -> API -> Tokens"
     exit 1
   fi

   echo "=== Installing Kubernetes Operators ==="

   # --- cert-manager ---
   echo ""
   echo "--- Installing cert-manager ---"
   helm repo add jetstack https://charts.jetstack.io 2>/dev/null || true
   helm repo update jetstack

   helm upgrade --install cert-manager jetstack/cert-manager \
     --namespace cert-manager \
     --create-namespace \
     --version v1.14.0 \
     --values "${SCRIPT_DIR}/cert-manager-values.yaml" \
     --wait \
     --timeout 5m \
     ${DRY_RUN}

   echo "cert-manager installed successfully"

   # Wait for cert-manager webhook to be ready
   if [[ -z "${DRY_RUN}" ]]; then
     echo "Waiting for cert-manager webhook..."
     kubectl wait --for=condition=Available deployment/cert-manager-webhook \
       -n cert-manager --timeout=120s
   fi

   # --- DigitalOcean credentials for external-dns and cert-manager ---
   echo ""
   echo "--- Creating DigitalOcean credentials secrets ---"

   # Create namespace if not exists
   kubectl create namespace nebula-dns ${DRY_RUN} 2>/dev/null || true

   # Secret for external-dns
   kubectl create secret generic digitalocean-credentials \
     --namespace kube-system \
     --from-literal=access-token="${DO_TOKEN}" \
     --dry-run=client -o yaml | kubectl apply ${DRY_RUN} -f -

   # Secret for cert-manager DNS-01 solver
   kubectl create secret generic digitalocean-dns \
     --namespace cert-manager \
     --from-literal=access-token="${DO_TOKEN}" \
     --dry-run=client -o yaml | kubectl apply ${DRY_RUN} -f -

   echo "Secrets created successfully"

   # --- external-dns ---
   echo ""
   echo "--- Installing external-dns ---"
   helm repo add external-dns https://kubernetes-sigs.github.io/external-dns/ 2>/dev/null || true
   helm repo update external-dns

   helm upgrade --install external-dns external-dns/external-dns \
     --namespace kube-system \
     --values "${SCRIPT_DIR}/external-dns-values.yaml" \
     --wait \
     --timeout 5m \
     ${DRY_RUN}

   echo "external-dns installed successfully"

   # --- Apply ClusterIssuer ---
   echo ""
   echo "--- Applying cert-manager ClusterIssuer ---"
   if [[ -z "${DRY_RUN}" ]]; then
     kubectl apply -f "${SCRIPT_DIR}/../../../k8s/base/cert-manager/cluster-issuer.yaml"
     echo "ClusterIssuer applied"
   else
     echo "(skipped in dry-run mode)"
   fi

   echo ""
   echo "=== All operators installed successfully ==="
   echo ""
   echo "Verify with:"
   echo "  kubectl get pods -n cert-manager"
   echo "  kubectl get pods -n kube-system -l app.kubernetes.io/name=external-dns"
   echo "  kubectl get clusterissuers"
   ```
   Make the script executable: `chmod +x install-operators.sh`
