  <action>
1. Create DOKS cluster module at `infrastructure/terraform/modules/doks-cluster/`:

   `main.tf`:
   ```hcl
   resource "digitalocean_kubernetes_cluster" "main" {
     name    = "${var.project_name}-${var.environment}"
     region  = var.region
     version = var.kubernetes_version
     vpc_uuid = var.vpc_uuid
     auto_upgrade  = var.auto_upgrade
     surge_upgrade = true
     ha            = var.ha_control_plane

     node_pool {
       name       = "${var.project_name}-${var.environment}-workers"
       size       = var.node_size
       node_count = var.node_count
       auto_scale = var.auto_scale
       min_nodes  = var.min_nodes
       max_nodes  = var.max_nodes
       labels = {
         environment = var.environment
         workload    = "general"
       }
     }

     maintenance_policy {
       day        = "sunday"
       start_time = "04:00"
     }
   }
   ```

   `variables.tf`: Variables for `project_name`, `environment`, `region`, `kubernetes_version` (default "1.30"), `vpc_uuid`, `node_size` (default "s-4vcpu-8gb"), `node_count` (default 3), `auto_scale` (default true), `min_nodes` (default 3), `max_nodes` (default 10), `auto_upgrade` (default true), `ha_control_plane` (default true).

   `outputs.tf`: Output `cluster_id`, `cluster_name`, `cluster_endpoint`, `cluster_kubeconfig` (sensitive, raw_config), `cluster_token` (sensitive).

2. Create production environment at `infrastructure/terraform/environments/production/`:

   `backend.tf`: Configure S3-compatible backend for DigitalOcean Spaces:
   ```hcl
   terraform {
     backend "s3" {
       bucket                      = "nebula-terraform-state"
       key                         = "production/terraform.tfstate"
       region                      = "us-east-1"  # Required but ignored by Spaces
       endpoint                    = "https://nyc3.digitaloceanspaces.com"
       skip_credentials_validation = true
       skip_metadata_api_check     = true
       skip_requesting_account_id  = true
       skip_s3_checksum            = true
     }
   }
   ```

   `variables.tf`: Variables for `do_token` (sensitive), `region` (default "nyc3"), `environment` (default "production").

   `main.tf`: Wire modules together:
   ```hcl
   terraform {
     required_version = ">= 1.7.0"
     required_providers {
       digitalocean = { source = "digitalocean/digitalocean", version = "~> 2.36" }
     }
   }

   provider "digitalocean" { token = var.do_token }

   module "vpc" {
     source       = "../../modules/vpc-network"
     project_name = "nebula-dns"
     environment  = var.environment
     region       = var.region
   }

   module "doks" {
     source             = "../../modules/doks-cluster"
     project_name       = "nebula-dns"
     environment        = var.environment
     region             = var.region
     vpc_uuid           = module.vpc.vpc_id
     kubernetes_version = "1.30"
     node_size          = "s-4vcpu-8gb"
     node_count         = 3
     min_nodes          = 3
     max_nodes          = 10
   }
   ```

   `outputs.tf`: Output `vpc_id`, `cluster_id`, `cluster_endpoint`, `kubeconfig` (sensitive).

   `terraform.tfvars.example`: Example values file showing all configurable vars (with placeholder `do_token = "dop_v1_..."`)

All files must include the copyright header as HCL comments.
