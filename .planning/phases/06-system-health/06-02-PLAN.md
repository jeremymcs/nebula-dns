  <action>
Update admin.controller.ts to add system health endpoints:

**Add imports:**
- AdminHealthService from './admin-health.service'
- SystemHealthResponseDto, ErrorRatesDto, QueryVolumeDto, StorageStatusDto, RecentErrorDto from './dto/system-health.dto'

**Add constructor injection:**
- Add `private readonly adminHealthService: AdminHealthService` to constructor

**Add endpoints (all protected by AdminAuthGuard):**

1. **GET /admin/system/health** - Combined system health
   ```typescript
   @Get('system/health')
   @UseGuards(AdminAuthGuard)
   @ApiBearerAuth('admin-jwt')
   @ApiOperation({
     summary: 'Get system health status',
     description: 'Returns combined health status of DNS services, storage connections, error rates, and query volume.',
   })
   @ApiResponse({
     status: 200,
     description: 'System health status',
     type: SystemHealthResponseDto,
   })
   async getSystemHealth(): Promise<SystemHealthResponseDto> {
     return this.adminHealthService.getSystemHealth();
   }
   ```

2. **GET /admin/system/errors** - Recent errors
   ```typescript
   @Get('system/errors')
   @UseGuards(AdminAuthGuard)
   @ApiBearerAuth('admin-jwt')
   @ApiOperation({
     summary: 'Get recent error entries',
     description: 'Returns recent DNS query errors from the log. Limit defaults to 20.',
   })
   @ApiQuery({ name: 'limit', required: false, type: Number, description: 'Max entries (default 20)' })
   @ApiResponse({
     status: 200,
     description: 'Recent error entries',
     type: [RecentErrorDto],
   })
   async getRecentErrors(
     @Query('limit', new DefaultValuePipe(20), ParseIntPipe) limit: number,
   ): Promise<RecentErrorDto[]> {
     return this.adminHealthService.getRecentErrors(limit);
   }
   ```

3. **GET /admin/system/query-volume** - Aggregate query volume
   ```typescript
   @Get('system/query-volume')
   @UseGuards(AdminAuthGuard)
   @ApiBearerAuth('admin-jwt')
   @ApiOperation({
     summary: 'Get aggregate query volume',
     description: 'Returns total query counts across all tenants for specified period.',
   })
   @ApiQuery({ name: 'period', required: false, enum: ['1h', '24h'], description: 'Time period (default 24h)' })
   @ApiResponse({
     status: 200,
     description: 'Aggregate query volume',
     type: QueryVolumeDto,
   })
   async getQueryVolume(
     @Query('period') period: '1h' | '24h' = '24h',
   ): Promise<QueryVolumeDto> {
     return this.adminHealthService.getQueryVolume(period);
   }
   ```

**Add required imports for new decorators:**
- DefaultValuePipe, ParseIntPipe from '@nestjs/common'
- ApiQuery from '@nestjs/swagger'

**Placement:**
- Add these endpoints AFTER the existing endpoints (login, search, MSP, impersonation)
- Keep grouped together as system health endpoints
