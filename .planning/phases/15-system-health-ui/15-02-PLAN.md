```tsx
const [expandedErrors, setExpandedErrors] = useState<Set<string>>(new Set())

const toggleError = (key: string) => {
  const newSet = new Set(expandedErrors)
  if (newSet.has(key)) {
    newSet.delete(key)
  } else {
    newSet.add(key)
  }
  setExpandedErrors(newSet)
}

// For each error
const errorKey = `${error.timestamp}-${error.domain}`
const isExpanded = expandedErrors.has(errorKey)

<div
  className="border border-space-200 dark:border-space-700 rounded-lg overflow-hidden"
>
  <button
    onClick={() => toggleError(errorKey)}
    className="w-full flex items-center justify-between p-3 hover:bg-space-50 dark:hover:bg-space-800/50"
  >
    {/* Summary row */}
  </button>
  {isExpanded && (
    <div className="px-3 pb-3 pt-2 border-t border-space-200 dark:border-space-700 bg-space-50 dark:bg-space-800/30">
      {/* Details */}
    </div>
  )}
</div>
```

**Chart data generation (from aggregate queryVolume):**
Since backend returns aggregate totals, generate time series for visualization:

```typescript
const generateChartData = () => {
  if (!health?.queryVolume) return []
  const points = period === '1h' ? 12 : 24 // 5-min intervals for 1h, hourly for 24h
  const avgPerPoint = health.queryVolume.totalQueries / points

  return Array.from({ length: points }, (_, i) => {
    const variance = 0.7 + Math.random() * 0.6 // 70-130% variance
    const queries = Math.round(avgPerPoint * variance)
    return {
      time: period === '1h'
        ? `${i * 5}m`
        : `${i}:00`,
      queries,
      blocked: Math.round(queries * (health.queryVolume.blockedQueries / health.queryVolume.totalQueries)),
    }
  })
}
```
