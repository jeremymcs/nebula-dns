**DOCR Cleanup Workflow:**

Create `.github/workflows/docr-cleanup.yml` to automatically clean up old container images.

```yaml
name: DOCR Cleanup

on:
  schedule:
    - cron: '0 4 * * 0'  # Weekly on Sunday at 4 AM UTC
  workflow_dispatch:  # Allow manual trigger

jobs:
  cleanup:
    name: Clean Up Old Images
    runs-on: ubuntu-latest
    steps:
      - name: Install doctl
        uses: digitalocean/action-doctl@v2
        with:
          token: ${{ secrets.DIGITALOCEAN_ACCESS_TOKEN }}

      - name: Start garbage collection
        run: |
          echo "## DOCR Garbage Collection" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY

          # Start garbage collection (removes untagged manifests)
          doctl registry garbage-collection start --force 2>&1 | tee gc-output.txt

          echo '```' >> $GITHUB_STEP_SUMMARY
          cat gc-output.txt >> $GITHUB_STEP_SUMMARY
          echo '```' >> $GITHUB_STEP_SUMMARY

      - name: Report registry usage
        run: |
          echo "" >> $GITHUB_STEP_SUMMARY
          echo "### Registry Repositories" >> $GITHUB_STEP_SUMMARY
          echo "" >> $GITHUB_STEP_SUMMARY

          doctl registry repository list-v2 --format Name,TagCount,ManifestCount 2>/dev/null >> $GITHUB_STEP_SUMMARY || echo "Could not list repositories" >> $GITHUB_STEP_SUMMARY
```
