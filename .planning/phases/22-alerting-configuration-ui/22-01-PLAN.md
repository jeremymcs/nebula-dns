  <files>admin-ui/src/lib/validations/alerting.schema.ts</files>
  <action>
    Create Zod validation schemas following admin-invite.schema.ts patterns:

    1. Import { z } from 'zod'

    2. Create SEVERITY_OPTIONS constant for form select:
       - { value: 'critical', label: 'Critical' }
       - { value: 'error', label: 'Error' }
       - { value: 'warning', label: 'Warning' }
       - { value: 'info', label: 'Info' }

    3. Create EVENT_TYPE_OPTIONS constant for form checkboxes:
       - { value: 'system_health', label: 'System Health' }
       - { value: 'billing', label: 'Billing' }
       - { value: 'security', label: 'Security' }

    4. Create slackConfigSchema:
       - name: z.string().min(1, 'Name is required').max(255, 'Name must be 255 characters or less')
       - webhookUrl: z.string()
           .min(1, 'Webhook URL is required')
           .url('Must be a valid URL')
           .regex(/^https:\/\/hooks\.slack\.com\//, 'Must start with https://hooks.slack.com/')
       - severityFilter: z.array(z.enum(['critical', 'error', 'warning', 'info'])).optional().default(['critical', 'error'])
       - eventTypes: z.array(z.enum(['system_health', 'billing', 'security'])).optional().default([])

    5. Create pagerDutyConfigSchema:
       - name: z.string().min(1, 'Name is required').max(255, 'Name must be 255 characters or less')
       - routingKey: z.string()
           .min(1, 'Routing key is required')
           .length(32, 'Routing key must be exactly 32 characters')
       - severityFilter: z.array(z.enum(['critical', 'error', 'warning', 'info'])).optional().default(['critical', 'error'])
       - eventTypes: z.array(z.enum(['system_health', 'billing', 'security'])).optional().default([])

    6. Create updateAlertConfigSchema (all optional except no credential changes):
       - name: z.string().min(1).max(255).optional()
       - isEnabled: z.boolean().optional()
       - severityFilter: z.array(z.enum(['critical', 'error', 'warning', 'info'])).optional()
       - eventTypes: z.array(z.enum(['system_health', 'billing', 'security'])).optional()

    7. Export inferred types:
       - SlackConfigFormData = z.infer<typeof slackConfigSchema>
       - PagerDutyConfigFormData = z.infer<typeof pagerDutyConfigSchema>
       - UpdateAlertConfigFormData = z.infer<typeof updateAlertConfigSchema>

    Include copyright header and file purpose comment.
