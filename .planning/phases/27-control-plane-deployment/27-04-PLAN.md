**Part A: Create Helm Ingress template**

Create `helm/safedns/templates/ingress.yaml` with dynamic host/TLS configuration from values:

```yaml
# Nebula DNS - Ingress (Helm Template)
# Routes HTTPS traffic to API, UI, and Admin UI services
# Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>

{{- if .Values.ingress.enabled }}
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: {{ include "safedns.fullname" . }}
  namespace: {{ .Release.Namespace }}
  labels:
    {{- include "safedns.labels" . | nindent 4 }}
  annotations:
    cert-manager.io/cluster-issuer: {{ .Values.ingress.clusterIssuer | default "letsencrypt-prod" }}
    nginx.ingress.kubernetes.io/ssl-redirect: "true"
    nginx.ingress.kubernetes.io/proxy-body-size: "10m"
    nginx.ingress.kubernetes.io/rate-limit-connections: "100"
    nginx.ingress.kubernetes.io/rate-limit-rps: "50"
    # CORS for API subdomain
    nginx.ingress.kubernetes.io/enable-cors: "true"
    nginx.ingress.kubernetes.io/cors-allow-origin: "https://{{ (index .Values.ingress.hosts 0).host }},https://{{ (index .Values.ingress.hosts 2).host }}"
    nginx.ingress.kubernetes.io/cors-allow-methods: "GET, POST, PUT, PATCH, DELETE, OPTIONS"
    nginx.ingress.kubernetes.io/cors-allow-headers: "Authorization, Content-Type, X-Requested-With"
    {{- with .Values.ingress.annotations }}
    {{- toYaml . | nindent 4 }}
    {{- end }}
spec:
  ingressClassName: {{ .Values.ingress.className | default "nginx" }}
  {{- if .Values.ingress.tls }}
  tls:
    {{- range .Values.ingress.tls }}
    - secretName: {{ .secretName }}
      hosts:
        {{- range .hosts }}
        - {{ . | quote }}
        {{- end }}
    {{- end }}
  {{- end }}
  rules:
    {{- range .Values.ingress.hosts }}
    - host: {{ .host | quote }}
      http:
        paths:
          {{- range .paths }}
          - path: {{ .path }}
            pathType: {{ .pathType }}
            backend:
              service:
                {{- if eq .service "api" }}
                name: {{ include "safedns.api.fullname" $ }}
                {{- else if eq .service "ui" }}
                name: {{ include "safedns.ui.fullname" $ }}
                {{- else if eq .service "admin-ui" }}
                name: {{ include "safedns.admin-ui.fullname" $ }}
                {{- else }}
                name: {{ .service }}
                {{- end }}
                port:
                  name: http
          {{- end }}
    {{- end }}
{{- end }}
```

**Part B: Create external-dns deployment**

Create `k8s/base/external-dns-deployment.yaml` with RBAC + Deployment for DigitalOcean DNS provider:

```yaml
# Nebula DNS - external-dns for DigitalOcean DNS Record Automation
# Watches Kubernetes Ingress/Service resources and creates DNS records
# Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>

apiVersion: v1
kind: ServiceAccount
metadata:
  name: external-dns
  namespace: kube-system
---
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRole
metadata:
  name: external-dns
rules:
  - apiGroups: [""]
    resources: ["services", "endpoints", "pods"]
    verbs: ["get", "watch", "list"]
  - apiGroups: ["extensions", "networking.k8s.io"]
    resources: ["ingresses"]
    verbs: ["get", "watch", "list"]
  - apiGroups: [""]
    resources: ["nodes"]
    verbs: ["list"]
---
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: external-dns-viewer
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: external-dns
subjects:
  - kind: ServiceAccount
    name: external-dns
    namespace: kube-system
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: external-dns
  namespace: kube-system
  labels:
    app: external-dns
    app.kubernetes.io/name: external-dns
    app.kubernetes.io/component: dns-automation
spec:
  replicas: 1
  selector:
    matchLabels:
      app: external-dns
  strategy:
    type: Recreate  # Prevent duplicate instances managing same DNS records
  template:
    metadata:
      labels:
        app: external-dns
    spec:
      serviceAccountName: external-dns
      securityContext:
        runAsNonRoot: true
        runAsUser: 65534  # nobody
        fsGroup: 65534
      containers:
        - name: external-dns
          image: registry.k8s.io/external-dns/external-dns:v0.14.0
          args:
            - --source=ingress
            - --source=service
            - --domain-filter=nebuladns.io
            - --provider=digitalocean
            - --policy=upsert-only  # Safer than sync: never deletes records
            - --txt-owner-id=nebula-dns-k8s
            - --txt-prefix=_externaldns.
            - --interval=1m
            - --log-level=info
          env:
            - name: DO_TOKEN
              valueFrom:
                secretKeyRef:
                  name: digitalocean-dns
                  key: access-token
          resources:
            requests:
              cpu: 10m
              memory: 32Mi
            limits:
              cpu: 50m
              memory: 64Mi
          securityContext:
            allowPrivilegeEscalation: false
            readOnlyRootFilesystem: true
            capabilities:
              drop:
                - ALL
```

Use `--policy=upsert-only` (not `sync`) to prevent accidental DNS record deletion during deployments. This is the safer option per Pitfall 4 in research.

**Part A: Create Kustomize Admin UI deployment**

Create `k8s/base/admin-ui-deployment.yaml` mirroring the UI deployment pattern from `k8s/base/ui-deployment.yaml`:

```yaml
# Nebula DNS - Admin UI Deployment
# Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>

apiVersion: apps/v1
kind: Deployment
metadata:
  name: admin-ui
  namespace: nebula-dns
  labels:
    app: admin-ui
    app.kubernetes.io/name: admin-ui
    app.kubernetes.io/component: frontend
spec:
  replicas: 2
  selector:
    matchLabels:
      app: admin-ui
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 1
      maxUnavailable: 0
  template:
    metadata:
      labels:
        app: admin-ui
    spec:
      securityContext:
        runAsNonRoot: true
        runAsUser: 1001
        fsGroup: 1001
      containers:
        - name: admin-ui
          image: ghcr.io/nebuladns/admin-ui:latest
          imagePullPolicy: Always
          ports:
            - name: http
              containerPort: 3000
              protocol: TCP
          env:
            - name: NODE_ENV
              value: "production"
            - name: PORT
              value: "3000"
            - name: HOSTNAME
              value: "0.0.0.0"
            - name: API_URL
              value: "http://api.nebula-dns.svc.cluster.local:80"
            - name: TENANT_UI_URL
              value: "https://app.nebuladns.io"
          resources:
            requests:
              cpu: 100m
              memory: 128Mi
            limits:
              cpu: 500m
              memory: 512Mi
          livenessProbe:
            httpGet:
              path: /api/config
              port: http
            initialDelaySeconds: 30
            periodSeconds: 10
            timeoutSeconds: 5
            failureThreshold: 3
          readinessProbe:
            httpGet:
              path: /api/config
              port: http
            initialDelaySeconds: 5
            periodSeconds: 5
            timeoutSeconds: 3
            failureThreshold: 3
          securityContext:
            allowPrivilegeEscalation: false
            readOnlyRootFilesystem: false
            capabilities:
              drop:
                - ALL
      affinity:
        podAntiAffinity:
          preferredDuringSchedulingIgnoredDuringExecution:
            - weight: 100
              podAffinityTerm:
                labelSelector:
                  matchLabels:
                    app: admin-ui
                topologyKey: kubernetes.io/hostname
---
apiVersion: v1
kind: Service
metadata:
  name: admin-ui
  namespace: nebula-dns
  labels:
    app: admin-ui
spec:
  type: ClusterIP
  ports:
    - port: 80
      targetPort: http
      protocol: TCP
      name: http
  selector:
    app: admin-ui
```

secrets:
