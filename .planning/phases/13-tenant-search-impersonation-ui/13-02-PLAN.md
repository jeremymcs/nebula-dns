/**
 * Nebula DNS - Impersonation Storage Helpers
 * Manage admin impersonation session data in localStorage
 * Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>
 */

const IMPERSONATION_TOKEN_KEY = 'impersonation_token'
const IMPERSONATION_INFO_KEY = 'impersonation_info'
const ADMIN_TOKEN_KEY = 'admin_token'
const TOKEN_KEY = 'token'

export interface ImpersonationInfo {
  tenantId: string
  tenantName: string
  tenantTier: 'home' | 'enterprise' | 'msp'
  startedAt: string
}

/**
 * Store impersonation session data
 * - Saves impersonation token separately (preserves admin_token)
 * - Sets main token for tenant dashboard to use
 */
export function storeImpersonation(
  token: string,
  info: Omit<ImpersonationInfo, 'startedAt'>
): void {
  localStorage.setItem(IMPERSONATION_TOKEN_KEY, token)
  localStorage.setItem(IMPERSONATION_INFO_KEY, JSON.stringify({
    ...info,
    startedAt: new Date().toISOString(),
  }))
  // Set as main token so tenant dashboard reads it
  localStorage.setItem(TOKEN_KEY, token)
}

/**
 * Get current impersonation info (null if not impersonating)
 */
export function getImpersonationInfo(): ImpersonationInfo | null {
  if (typeof window === 'undefined') return null
  const info = localStorage.getItem(IMPERSONATION_INFO_KEY)
  return info ? JSON.parse(info) : null
}

/**
 * Check if currently in impersonation mode
 */
export function isImpersonating(): boolean {
  if (typeof window === 'undefined') return false
  return !!localStorage.getItem(IMPERSONATION_TOKEN_KEY)
}

/**
 * Get impersonation token (for API calls during end)
 */
export function getImpersonationToken(): string | null {
  if (typeof window === 'undefined') return null
  return localStorage.getItem(IMPERSONATION_TOKEN_KEY)
}

/**
 * Clear all impersonation data
 * - Removes impersonation token and info
 * - Clears main token (admin_token remains)
 */
export function clearImpersonation(): void {
  localStorage.removeItem(IMPERSONATION_TOKEN_KEY)
  localStorage.removeItem(IMPERSONATION_INFO_KEY)
  localStorage.removeItem(TOKEN_KEY)
  // Note: admin_token is preserved
}
```

/**
 * Nebula DNS - Impersonation Hook
 * React hook for managing impersonation state
 * Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>
 */

'use client'

import { useState, useEffect, useCallback } from 'react'
import {
  ImpersonationInfo,
  getImpersonationInfo,
  getImpersonationToken,
  clearImpersonation,
  isImpersonating as checkIsImpersonating,
} from '@/lib/impersonation'

const API_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:4000'

/**
 * End impersonation session via API
 * Blacklists the impersonation token on server
 */
async function endImpersonationApi(): Promise<void> {
  const token = getImpersonationToken()
  if (!token) return

  try {
    await fetch(`${API_URL}/admin/impersonate/end`, {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${token}`,
      },
    })
  } catch {
    // Continue with cleanup even if API fails
    console.warn('Failed to end impersonation via API')
  }
}

export function useImpersonation() {
  const [impersonationInfo, setImpersonationInfo] = useState<ImpersonationInfo | null>(null)
  const [isImpersonating, setIsImpersonating] = useState(false)

  // Read impersonation state on mount and storage changes
  useEffect(() => {
    const updateState = () => {
      setImpersonationInfo(getImpersonationInfo())
      setIsImpersonating(checkIsImpersonating())
    }

    updateState()

    // Listen for storage changes (e.g., from other tabs)
    window.addEventListener('storage', updateState)
    return () => window.removeEventListener('storage', updateState)
  }, [])

  /**
   * End impersonation session
   * - Calls API to blacklist token
   * - Clears local storage
   * - Redirects to Mission Control
   */
  const endImpersonation = useCallback(async () => {
    await endImpersonationApi()
    clearImpersonation()
    window.location.href = '/mission-control'
  }, [])

  return {
    impersonationInfo,
    isImpersonating,
    endImpersonation,
  }
}

/**
 * Nebula DNS - Impersonation Banner Component
 * Persistent amber banner shown when admin is viewing as tenant
 * Copyright (c) 2026 Jeremy McSpadden <jeremy@fluxlabs.net>
 */

'use client'

import { EyeIcon } from '@heroicons/react/24/outline'
import { ImpersonationInfo } from '@/lib/impersonation'

interface ImpersonationBannerProps {
  impersonationInfo: ImpersonationInfo
  onEndImpersonation: () => void
}

/**
 * Get display label for tenant tier
 */
function getTierLabel(tier: ImpersonationInfo['tenantTier']): string {
  switch (tier) {
    case 'msp':
      return 'MSP'
    case 'enterprise':
      return 'Business'
    case 'home':
      return 'Home'
    default:
      return tier
  }
}

/**
 * ImpersonationBanner - Always visible during impersonation
 *
 * Per CONTEXT.md:
 * - Full-width amber/yellow banner at top of screen
 * - Always visible - cannot be dismissed during impersonation
 * - Shows: "Viewing as: [Tenant Name] (MSP/Business/Home)" with End Impersonation button
 */
export function ImpersonationBanner({
  impersonationInfo,
  onEndImpersonation,
}: ImpersonationBannerProps) {
  const tierLabel = getTierLabel(impersonationInfo.tenantTier)

  return (
    <div
      className="bg-amber-500 text-white px-4 py-2 flex items-center justify-between"
      role="alert"
      aria-live="polite"
    >
      <span className="text-sm font-medium flex items-center gap-2">
        <EyeIcon className="h-4 w-4 flex-shrink-0" />
        <span>
          Viewing as: <strong>{impersonationInfo.tenantName}</strong> ({tierLabel})
        </span>
      </span>
      <button
        onClick={onEndImpersonation}
        className="text-sm bg-white/20 hover:bg-white/30 px-3 py-1 rounded transition-colors font-medium"
      >
        End Impersonation
      </button>
    </div>
  )
}

export default ImpersonationBanner
```

Design notes (per CONTEXT.md):
- Amber/yellow background (bg-amber-500)
- White text for contrast
- Eye icon to indicate "viewing as"
- Cannot be dismissed (no X button, no dismiss state)
- End Impersonation button on right
- Follows GracePeriodBanner structure pattern
