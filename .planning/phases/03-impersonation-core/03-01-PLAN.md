```typescript
async startImpersonation(
  adminUser: AdminUser,  // From request.user
  tenantId: string,
  ipAddress?: string,
): Promise<StartImpersonationResponseDto> {
  // 1. Verify tenant exists
  const tenant = await this.tenantRepository.findOne({
    where: { id: tenantId },
  });

  if (!tenant) {
    throw new NotFoundException('Tenant not found');
  }

  // 2. Generate impersonation token with tenant JWT_SECRET
  const jti = uuidv4();
  const now = Math.floor(Date.now() / 1000);

  const payload = {
    sub: adminUser.adminId,      // Admin ID as subject
    email: adminUser.email,
    tenantId: tenant.id,         // Target tenant context
    role: 'admin',               // Full tenant admin access
    jti,
    mfaVerified: true,           // Admin already MFA verified
    // Impersonation-specific claims
    impersonatingAdminId: adminUser.adminId,
    impersonatingAdminEmail: adminUser.email,
    impersonationStartTime: now,
  };

  // Sign with tenant secret (JWT_SECRET, NOT ADMIN_JWT_SECRET)
  // Use 1-hour expiry for impersonation (shorter than admin's 8h)
  const accessToken = this.jwtService.sign(payload, {
    secret: this.configService.get<string>('JWT_SECRET'),
    expiresIn: '1h',
    issuer: 'nebula-dns',  // Tenant issuer, not nebula-admin
  });

  // 3. Audit log the start event
  await this.auditLogsService.log({
    tenantId: tenant.id,
    userId: null,  // No tenant user - it's an admin action
    action: 'admin_impersonation_start' as AuditAction,
    resourceType: 'tenant' as AuditResourceType,
    resourceId: tenant.id,
    resourceName: tenant.name,
    ipAddress: ipAddress || null,
    metadata: {
      adminId: adminUser.adminId,
      adminEmail: adminUser.email,
      tenantName: tenant.name,
      tenantTier: tenant.tier,
    },
  });

  this.logger.log(
    `Admin ${adminUser.email} started impersonating tenant ${tenant.name} (${tenant.id})`,
  );

  return {
    accessToken,
    tenantId: tenant.id,
    tenantName: tenant.name,
    tenantTier: tenant.tier,
    expiresIn: 3600,  // 1 hour in seconds
  };
}
```

4. Implement `endImpersonation` method:
```typescript
async endImpersonation(
  impersonationToken: string,
  ipAddress?: string,
): Promise<EndImpersonationResponseDto> {
  // 1. Verify and decode the impersonation token
  let payload: any;
  try {
    payload = this.jwtService.verify(impersonationToken, {
      secret: this.configService.get<string>('JWT_SECRET'),
    });
  } catch (error) {
    throw new UnauthorizedException('Invalid or expired impersonation token');
  }

  // 2. Validate it's an impersonation token (has impersonatingAdminId claim)
  if (!payload.impersonatingAdminId) {
    throw new BadRequestException('Not an impersonation token');
  }

  // 3. Blacklist the token using session service
  // Note: Use adminSessionService for blacklisting since it has the method
  await this.adminSessionService.blacklistToken(payload.jti);

  // 4. Calculate session duration
  const startTime = payload.impersonationStartTime || 0;
  const endTime = Math.floor(Date.now() / 1000);
  const durationSeconds = endTime - startTime;

  // 5. Audit log the end event
  await this.auditLogsService.log({
    tenantId: payload.tenantId,
    userId: null,
    action: 'admin_impersonation_end' as AuditAction,
    resourceType: 'tenant' as AuditResourceType,
    resourceId: payload.tenantId,
    ipAddress: ipAddress || null,
    metadata: {
      adminId: payload.impersonatingAdminId,
      adminEmail: payload.impersonatingAdminEmail,
      durationSeconds,
      startTime: new Date(startTime * 1000).toISOString(),
      endTime: new Date(endTime * 1000).toISOString(),
    },
  });

  this.logger.log(
    `Admin ${payload.impersonatingAdminEmail} ended impersonation of tenant ${payload.tenantId} after ${durationSeconds}s`,
  );

  return {
    message: 'Impersonation session ended',
    durationSeconds,
  };
}
```

/**
 * Start impersonation session
 * Generates a tenant-scoped JWT for accessing tenant dashboard
 */
@Post('impersonate/:tenantId')
@UseGuards(AdminAuthGuard)
@ApiBearerAuth('admin-jwt')
@ApiOperation({
  summary: 'Start impersonation session',
  description:
    'Generate a JWT token to access the target tenant\'s dashboard as admin. ' +
    'The token is scoped to the tenant and grants full admin access. ' +
    'Impersonation sessions are limited to 1 hour and are fully audit-logged.',
})
@ApiResponse({
  status: 200,
  description: 'Impersonation token generated successfully',
  type: StartImpersonationResponseDto,
})
@ApiResponse({
  status: 404,
  description: 'Tenant not found',
})
@ApiResponse({
  status: 401,
  description: 'Admin authentication required',
})
async startImpersonation(
  @Param('tenantId') tenantId: string,
  @Req() req: Request,
  @Ip() ip: string,
): Promise<StartImpersonationResponseDto> {
  const adminUser = req.user as AdminUser;
  return this.adminService.startImpersonation(
    adminUser,
    tenantId,
    this.getClientIp(req, ip),
  );
}
```

/**
 * End impersonation session
 * Blacklists the impersonation token and logs the end event
 */
@Post('impersonate/end')
@HttpCode(HttpStatus.OK)
@ApiBearerAuth()  // Uses the impersonation token (tenant JWT), not admin JWT
@ApiOperation({
  summary: 'End impersonation session',
  description:
    'End the current impersonation session by blacklisting the impersonation token. ' +
    'The admin should then use their original admin token to continue. ' +
    'Returns the duration of the impersonation session.',
})
@ApiResponse({
  status: 200,
  description: 'Impersonation session ended',
  type: EndImpersonationResponseDto,
})
@ApiResponse({
  status: 400,
  description: 'Not an impersonation token',
})
@ApiResponse({
  status: 401,
  description: 'Invalid or expired token',
})
async endImpersonation(
  @Req() req: Request,
  @Ip() ip: string,
): Promise<EndImpersonationResponseDto> {
  // Extract token from Authorization header
  const authHeader = req.headers.authorization;
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    throw new UnauthorizedException('No token provided');
  }
  const token = authHeader.replace('Bearer ', '');
  return this.adminService.endImpersonation(token, this.getClientIp(req, ip));
}
```

Important notes:
- Start impersonation requires AdminAuthGuard (admin JWT)
- End impersonation does NOT use AdminAuthGuard - it accepts the impersonation token directly
- The /impersonate/end route must be placed BEFORE any :param routes to avoid route conflicts
