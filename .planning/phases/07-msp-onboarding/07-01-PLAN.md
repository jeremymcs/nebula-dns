   ```typescript
   private async generateUniqueSlug(name: string): Promise<string> {
     const baseSlug = name
       .toLowerCase()
       .replace(/[^a-z0-9]+/g, '-')
       .replace(/(^-|-$)/g, '');

     let slug = baseSlug;
     let counter = 1;
     while (await this.tenantRepository.findOne({ where: { slug } })) {
       slug = `${baseSlug}-${counter}`;
       counter++;
     }
     return slug;
   }
   ```
