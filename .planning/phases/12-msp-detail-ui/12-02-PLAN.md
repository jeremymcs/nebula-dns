```typescript
export interface UseAdminMspClientsParams {
  q?: string
  status?: string
  limit?: number
  cursor?: string
  period?: '24h' | '7d' | '30d' | '90d'
}

export function useAdminMspClients(mspId: string, params?: UseAdminMspClientsParams) {
  const searchParams = new URLSearchParams()
  if (params?.q) searchParams.set('q', params.q)
  if (params?.status) searchParams.set('status', params.status)
  if (params?.limit) searchParams.set('limit', params.limit.toString())
  if (params?.cursor) searchParams.set('cursor', params.cursor)
  if (params?.period) searchParams.set('period', params.period)
  const query = searchParams.toString()
  const endpoint = `/admin/msp/tenants/${mspId}/clients${query ? `?${query}` : ''}`

  return useSWR<MspClientListResponse>(endpoint, adminFetcher, defaultConfig)
}
```

8. Generate mock chart data from usage metrics (API returns aggregates, not time series):
```typescript
const generateChartData = () => {
  if (!msp?.usage) return []
  // Generate simple mock time series from aggregate data
  const days = period === '24h' ? 24 : period === '7d' ? 7 : 30
  const avgQueries = Math.round(msp.usage.totalQueries / days)
  const avgBlocked = Math.round(msp.usage.blockedQueries / days)

  return Array.from({ length: days }, (_, i) => {
    const variance = 0.8 + Math.random() * 0.4 // 80-120% variance
    return {
      date: period === '24h'
        ? `${i}:00`
        : new Date(Date.now() - (days - 1 - i) * 24 * 60 * 60 * 1000).toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
      queries: Math.round(avgQueries * variance),
      blocked: Math.round(avgBlocked * variance),
    }
  })
}
const chartData = generateChartData()
```
