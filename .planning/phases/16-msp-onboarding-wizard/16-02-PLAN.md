2. Add localStorage draft persistence:
```typescript
const STORAGE_KEY = 'msp-onboarding-draft'
const DRAFT_EXPIRY_MS = 24 * 60 * 60 * 1000 // 24 hours

// On mount: Check for existing draft
useEffect(() => {
  const saved = localStorage.getItem(STORAGE_KEY)
  if (saved) {
    try {
      const { formData, step, savedAt } = JSON.parse(saved)
      if (Date.now() - savedAt < DRAFT_EXPIRY_MS) {
        // Show toast asking to resume?
        // For now, auto-restore
        reset(formData)
        setCurrentStep(step)
      } else {
        localStorage.removeItem(STORAGE_KEY)
      }
    } catch {
      localStorage.removeItem(STORAGE_KEY)
    }
  }
}, [])

// On form change: Save draft (debounced)
const watchedValues = watch()
useEffect(() => {
  const timeoutId = setTimeout(() => {
    if (watchedValues.name) { // Only save if meaningful data
      localStorage.setItem(STORAGE_KEY, JSON.stringify({
        formData: watchedValues,
        step: currentStep,
        savedAt: Date.now(),
      }))
    }
  }, 500) // Debounce 500ms
  return () => clearTimeout(timeoutId)
}, [watchedValues, currentStep])
```

4. Implement form submission:
```typescript
const handleSubmit = async () => {
  setIsSubmitting(true)
  try {
    const formData = getValues()
    const data: CreateMspTenantDto = {
      name: formData.name,
      slug: formData.slug || undefined,
      adminEmail: formData.adminEmail,
      adminName: formData.adminName || undefined,
      contactEmail: formData.contactEmail || undefined,
      contactPhone: formData.contactPhone || undefined,
      locationLimit: formData.locationLimit ?? undefined,
      clientLimit: formData.clientLimit ?? undefined,
    }
    const result = await createMspTenant(data)
    localStorage.removeItem(STORAGE_KEY) // Clear draft on success
    setCreatedMsp(result)
    setIsComplete(true)
  } catch (error) {
    showToast('error', error instanceof Error ? error.message : 'Failed to create MSP')
  } finally {
    setIsSubmitting(false)
  }
}
```
