export const mspCompanyInfoSchema = z.object({
  name: z.string()
    .min(2, 'Name must be at least 2 characters')
    .max(255, 'Name must not exceed 255 characters'),
  slug: z.string()
    .regex(/^[a-z0-9-]+$/, 'Slug must contain only lowercase letters, numbers, and hyphens')
    .min(2, 'Slug must be at least 2 characters')
    .max(100, 'Slug must not exceed 100 characters')
    .optional()
    .or(z.literal('')),
  contactEmail: z.string().email('Invalid email').optional().or(z.literal('')),
  contactPhone: z.string().max(50).optional().or(z.literal('')),
})

// Step 2: Admin User
export const mspAdminUserSchema = z.object({
  adminEmail: z.string().email('Admin email must be a valid email'),
  adminName: z.string()
    .min(2, 'Name must be at least 2 characters')
    .max(255, 'Name must not exceed 255 characters')
    .optional()
    .or(z.literal('')),
})

// Step 3: Settings
export const mspSettingsSchema = z.object({
  locationLimit: z.number()
    .int()
    .min(1)
    .max(10000)
    .optional()
    .nullable(),
  clientLimit: z.number()
    .int()
    .min(1)
    .max(10000)
    .optional()
    .nullable(),

// MSP Creation Types
export interface CreateMspTenantDto {
  name: string
  slug?: string
  adminEmail: string
  adminName?: string
  contactEmail?: string
  contactPhone?: string
  locationLimit?: number | null
  clientLimit?: number | null
}

export interface MspTenantCreatedResponse {
  id: string
  name: string
  slug: string
  status: 'active'
  tier: 'msp'
  adminUserId: string
  adminEmail: string
  welcomeEmailSent: boolean
  locationLimit: number | null
  clientLimit: number | null
  createdAt: string
}

export async function createMspTenant(data: CreateMspTenantDto): Promise<MspTenantCreatedResponse> {
  const token = typeof window !== 'undefined' ? localStorage.getItem('admin_token') : null
  if (!token) throw new Error('Admin authentication required')

  const res = await fetch(`${API_URL}/admin/msp/tenants`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${token}`,
    },
    body: JSON.stringify(data),
  })

  if (!res.ok) {
    const error = await res.json().catch(() => ({ message: 'Request failed' }))
    throw new Error(error.message || `Request failed (${res.status})`)
  }

  return res.json()
}
