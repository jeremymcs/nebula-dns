   - @InjectRepository(User) private readonly userRepository: Repository<User>
   - Add imports for Location, User entities and new DTOs

2. **Add verifyMspTenant helper** (private):
   ```typescript
   private async verifyMspTenant(mspId: string): Promise<Tenant> {
     const tenant = await this.tenantRepository.findOne({
       where: { id: mspId },
     });
     if (!tenant) {
       throw new NotFoundException('Tenant not found');
     }
     if (tenant.tier !== 'msp') {
       throw new BadRequestException('Tenant is not an MSP');
     }
     return tenant;
   }
   ```

3. **Add getMspDetail method**:
   - Accept mspId: string, query: MspDetailQueryDto
   - Call verifyMspTenant(mspId)
   - Count clients: COUNT from tenants WHERE managed_by_msp_id = mspId
   - Count locations: JOIN locations.tenant WHERE tenant.managed_by_msp_id = mspId
   - Count users: JOIN users.tenant WHERE tenant.managed_by_msp_id = mspId
   - Return MspDetailResponseDto with all fields

4. **Add getMspClients method**:
   - Accept mspId: string, query: MspClientListQueryDto
   - Call verifyMspTenant(mspId)
   - Use QueryBuilder pattern from listMspTenants (LEFT JOIN for counts)
   - Query clients with location/user counts via aggregation:
     ```typescript
     const queryBuilder = this.tenantRepository
       .createQueryBuilder('client')
       .leftJoin('client.locations', 'location')
       .leftJoin('client.users', 'user')
       .select([
         'client.id AS id',
         'client.name AS name',
         'client.status AS status',
         'client.tier AS tier',
         'client.contact_email AS "contactEmail"',
         'client.created_at AS "createdAt"',
       ])
       .addSelect('COUNT(DISTINCT location.id)', 'locationCount')
       .addSelect('COUNT(DISTINCT user.id)', 'userCount')
       .where('client.managed_by_msp_id = :mspId', { mspId })
       .groupBy('client.id');
     ```
   - Apply search filter if query.q (name ILIKE)
   - Apply status filter if query.status
   - Get total count with same filters
   - Apply pagination (offset/limit)
   - Return MspClientListResponseDto

2. **Import ParseUUIDPipe** from @nestjs/common for mspId validation

3. **Add GET /msp/tenants/:mspId endpoint**:
   ```typescript
   @Get('msp/tenants/:mspId')
   @UseGuards(AdminAuthGuard)
   @ApiBearerAuth('admin-jwt')
   @ApiOperation({
     summary: 'Get MSP tenant details',
     description: 'Get detailed information about an MSP including client, location, and user counts.',
   })
   @ApiParam({ name: 'mspId', description: 'MSP tenant UUID', type: 'string' })
   @ApiResponse({ status: 200, type: MspDetailResponseDto, description: 'MSP details retrieved' })
   @ApiResponse({ status: 400, description: 'Tenant is not an MSP' })
   @ApiResponse({ status: 404, description: 'Tenant not found' })
   async getMspDetail(
     @Param('mspId', ParseUUIDPipe) mspId: string,
     @Query() query: MspDetailQueryDto,
   ): Promise<MspDetailResponseDto> {
     return this.adminService.getMspDetail(mspId, query);
   }
   ```

4. **Add GET /msp/tenants/:mspId/clients endpoint**:
   ```typescript
   @Get('msp/tenants/:mspId/clients')
   @UseGuards(AdminAuthGuard)
   @ApiBearerAuth('admin-jwt')
   @ApiOperation({
     summary: 'List MSP client organizations',
     description: 'List all client organizations managed by an MSP with location and user counts.',
   })
   @ApiParam({ name: 'mspId', description: 'MSP tenant UUID', type: 'string' })
   @ApiResponse({ status: 200, type: MspClientListResponseDto, description: 'Client list retrieved' })
   @ApiResponse({ status: 400, description: 'Tenant is not an MSP' })
   @ApiResponse({ status: 404, description: 'Tenant not found' })
   async getMspClients(
     @Param('mspId', ParseUUIDPipe) mspId: string,
     @Query() query: MspClientListQueryDto,
   ): Promise<MspClientListResponseDto> {
     return this.adminService.getMspClients(mspId, query);
   }
   ```
