   ```typescript
   private getPeriodDates(period: string): { startDate: Date; endDate: Date } {
     const now = new Date();
     let startDate: Date;

     switch (period) {
       case '24h':
         startDate = new Date(now.getTime() - 24 * 60 * 60 * 1000);
         break;
       case '7d':
         startDate = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
         break;
       case '90d':
         startDate = new Date(now.getTime() - 90 * 24 * 60 * 60 * 1000);
         break;
       case '30d':
       default:
         startDate = new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000);
     }

     return { startDate, endDate: now };
   }
   ```

3. **Add aggregateUsageForMsp method** (private):
   ```typescript
   private async aggregateUsageForMsp(
     mspId: string,
     period: '24h' | '7d' | '30d' | '90d' = '30d',
   ): Promise<MspUsageMetricsDto> {
     // Get all client tenant IDs
     const clients = await this.tenantRepository.find({
       where: { managedByMspId: mspId },
       select: ['id'],
     });

     const { startDate, endDate } = this.getPeriodDates(period);
     const isMockData = this.clickhouseService.isUsingMockData();

     let totalQueries = 0;
     let blockedQueries = 0;
     let threatsBlocked = 0;

     // Aggregate across all clients
     for (const client of clients) {
       try {
         const stats = await this.clickhouseService.getSummaryStats(client.id, period);
         totalQueries += Number(stats.totalQueries) || 0;
         blockedQueries += Number(stats.blockedQueries) || 0;
         threatsBlocked += Number(stats.threatsBlocked) || 0;
       } catch (error) {
         this.logger.warn(`Failed to get stats for client ${client.id}: ${(error as Error).message}`);
       }
     }

     return {
       totalQueries,
       blockedQueries,
       threatsBlocked,
       blockRate: totalQueries > 0
         ? Math.round((blockedQueries / totalQueries) * 1000) / 10
         : 0,
       periodStart: startDate.toISOString(),
       periodEnd: endDate.toISOString(),
       isMockData,
     };
   }
   ```

4. **Add getClientUsage method** (private):

   ```typescript
     try {

       return {

         blockedQueries: Number(stats.blockedQueries) || 0,
       };
     } catch (error) {
       this.logger.warn(`Failed to get usage for tenant ${tenantId}`);
       return { totalQueries: 0, blockedQueries: 0 };
     }
   }
   ```
